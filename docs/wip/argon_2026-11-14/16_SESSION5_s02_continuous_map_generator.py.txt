# 2. CAMERA & PROJECTION MATHS (LOG-SCALE ZOOM + CUBIC EASING)
# =============================================================================

@dataclass(frozen=True)
class View:
    lon: float
    lat: float
    span_km: float
    width: int = WIDTH
    height: int = HEIGHT
    pitch_deg: float = 0.0

    @property
    def km_per_px(self) -> float:
        return self.span_km / self.width

    @property
    def height_km(self) -> float:
        return self.span_km * self.height / self.width

    def to_px(self, lon, lat):
        lon_a = np.asarray(lon, dtype=np.float64)
        lat_a = np.asarray(lat, dtype=np.float64)
        kx = KM_PER_DEG * math.cos(math.radians(self.lat))
        x = ((lon_a - self.lon) * kx / self.span_km + 0.5) * self.width
        y = (0.5 - (lat_a - self.lat) * KM_PER_DEG / self.height_km) * self.height
        return x, y

    def to_lonlat(self, x, y):
        x_a = np.asarray(x, dtype=np.float64)
        y_a = np.asarray(y, dtype=np.float64)
        kx = KM_PER_DEG * math.cos(math.radians(self.lat))
        lon = self.lon + (x_a / self.width - 0.5) * self.span_km / kx
        lat = self.lat + (0.5 - y_a / self.height) * self.height_km / KM_PER_DEG
        return lon, lat


def ease_in_out(u: float) -> float:
    u = min(max(float(u), 0.0), 1.0)
    return 4.0 * u ** 3 if u < 0.5 else 1.0 - (-2.0 * u + 2.0) ** 3 / 2.0


def march_ease(u: float, lead: float = 0.14, tail: float = 0.14) -> float:
    u = min(max(float(u), 0.0), 1.0)
    v = 1.0 / (1.0 - (lead + tail) / 2.0)
    if u < lead:
        return v * u * u / (2.0 * lead)
    if u > 1.0 - tail:
        r = 1.0 - u
        return 1.0 - v * r * r / (2.0 * tail)
    return v * lead / 2.0 + v * (u - lead)


def lerp_view(v0: View, v1: View, u: float) -> View:
    """Interpolate camera with position compensated for log-scale span_km (Law 5)."""
    e = ease_in_out(u)
    log_s0, log_s1 = math.log(v0.span_km), math.log(v1.span_km)
    span = math.exp(log_s0 + (log_s1 - log_s0) * e)
    if abs(v1.span_km - v0.span_km) > 25.0:
        pan_u = (span - v0.span_km) / (v1.span_km - v0.span_km)
        pan_u = 0.65 * pan_u + 0.35 * e
    else:
        pan_u = e
    lon = v0.lon + (v1.lon - v0.lon) * pan_u
    lat = v0.lat + (v1.lat - v0.lat) * pan_u
    pitch = v0.pitch_deg + (v1.pitch_deg - v0.pitch_deg) * e
    return View(lon, lat, span, v0.width, v0.height, pitch)


def camera_at_frame(f: int) -> View:
    """Continuous 453-frame camera trajectory across S02_SH001 -> S02_SH002 -> S02_SH003."""
    kf_start = View(4.80, 50.08, 620.0, pitch_deg=5.0)       # f=0: Belgium & N. France
    kf_aug = View(4.35, 49.85, 660.0, pitch_deg=6.0)         # f=125: Aachen -> Brussels -> Marne in frame
    kf_dec_wide = View(4.92, 49.38, 950.0, pitch_deg=6.0)    # f=215: Full North Sea to Swiss border theater
    kf_dec_end = View(4.95, 49.34, 940.0, pitch_deg=7.0)     # f=325: End of trench draw
    kf_flanders = View(2.89, 50.70, 125.0, pitch_deg=11.0)   # f=398: Flanders sector (Ypres -> La Bassee)
    kf_close = View(2.895, 50.705, 102.0, pitch_deg=12.0)    # f=452: Frelinghien-Ploegsteert push

    if f <= 125:
        return lerp_view(kf_start, kf_aug, f / 125.0)
    if f <= 215:
        return lerp_view(kf_aug, kf_dec_wide, (f - 125) / 90.0)
    if f <= 325:
        return lerp_view(kf_dec_wide, kf_dec_end, (f - 215) / 110.0)
    if f <= 398:
        return lerp_view(kf_dec_end, kf_flanders, (f - 325) / 73.0)
    return lerp_view(kf_flanders, kf_close, (f - 398) / max(1, (TOTAL_FRAMES - 1 - 398)))

def catmull_rom_spline(waypoints: Sequence[tuple[float, float]], per_seg: int = 28) -> list[tuple[float, float]]:
    pts = [tuple(p) for p in waypoints]
    if len(pts) < 3:
        return [
            (pts[0][0] + (pts[-1][0] - pts[0][0]) * k / per_seg,
             pts[0][1] + (pts[-1][1] - pts[0][1]) * k / per_seg)
            for k in range(per_seg + 1)
        ]
    ext = [pts[0]] + pts + [pts[-1]]
    out: list[tuple[float, float]] = []
    for i in range(1, len(ext) - 2):
        p0, p1, p2, p3 = ext[i - 1], ext[i], ext[i + 1], ext[i + 2]
        for k in range(per_seg):
            t = k / per_seg
            t2, t3 = t * t, t * t * t
            x = 0.5 * ((2 * p1[0]) + (-p0[0] + p2[0]) * t + (2 * p0[0] - 5 * p1[0] + 4 * p2[0] - p3[0]) * t2 + (-p0[0] + 3 * p1[0] - 3 * p2[0] + p3[0]) * t3)
            y = 0.5 * ((2 * p1[1]) + (-p0[1] + p2[1]) * t + (2 * p0[1] - 5 * p1[1] + 4 * p2[1] - p3[1]) * t2 + (-p0[1] + 3 * p1[1] - 3 * p2[1] + p3[1]) * t3)
            out.append((x, y))
    out.append(pts[-1])
    return out


def polyline_prefix(pts: list[tuple[float, float]], share: float) -> list[tuple[float, float]]:
    if share <= 0.0 or len(pts) < 2:
        return []
    if share >= 1.0:
        return list(pts)
    lens = [math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(pts, pts[1:])]
    goal = sum(lens) * share
    out = [pts[0]]
    run = 0.0
    for (a, b), seg_len in zip(zip(pts, pts[1:]), lens):
        if run + seg_len >= goal and seg_len > 1e-6:
            t = (goal - run) / seg_len
            out.append((a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t))
            return out
        out.append(b)
        run += seg_len
    return out


def place_label_8pos(

    th: float,
    occupied: list[tuple[float, float, float, float]],

    pad = max(8.0, th * 0.38)

            (-pad - tw, -pad - th, False),

            (-pad * 1.4 - tw, -th * 0.5, False),
            (+pad * 2.4, -pad * 2.0 - th, True),
            (-pad * 2.4 - tw, -pad * 2.0 - th, True),

        "NW": [
            (-pad - tw, -pad - th, False),
            (+pad, -pad - th, False),
            (-pad - tw, +pad, False),
            (+pad, +pad, False),
            (-pad * 1.4 - tw, -th * 0.5, False),
            (+pad * 1.4, -th * 0.5, False),
            (-pad * 2.4 - tw, -pad * 2.0 - th, True),
            (+pad * 2.4, -pad * 2.0 - th, True),
        ],
        "E": [
            (+pad * 1.35, -th * 0.5, False),
            (+pad, -pad - th, False),
            (+pad, +pad, False),
            (-pad * 1.35 - tw, -th * 0.5, False),
            (-pad - tw, -pad - th, False),
            (-pad - tw, +pad, False),
            (+pad * 2.5, -th * 0.5, True),

        "W": [

            (+pad, -pad - th, False),
            (-pad - tw, +pad, False),
            (-pad - tw, -pad - th, False),
            (+pad * 1.35, -th * 0.5, False),
            (-pad * 1.35 - tw, -th * 0.5, False),
            (+pad * 2.4, +pad * 2.0, True),
            (-pad * 2.4 - tw, +pad * 2.0, True),
        ],
    }
    candidates = order_map.get(prefer, order_map["NE"])
    xmin, xmax = 54.0, WIDTH - 54.0
    ymin, ymax = 40.0, HEIGHT - 66.0

    best = None
    best_cost = float("inf")
    for idx, (dx, dy, leader) in enumerate(candidates):
        x0 = float(np.clip(cx + dx, xmin, xmax - tw))
        y0 = float(np.clip(cy + dy, ymin, ymax - th))
        x1, y1 = x0 + tw, y0 + th

        overlap = 0.0
        for ox0, oy0, ox1, oy1 in occupied:
            iw = max(0.0, min(x1, ox1) - max(x0, ox0))
            ih = max(0.0, min(y1, oy1) - max(y0, oy0))
            overlap += iw * ih

        cost = overlap * 1000.0 + idx * 8.0 + (30.0 if leader else 0.0)
        if cost < best_cost:
            best_cost = cost
            best = (x0, y0, (x0 - 6.0, y0 - 4.0, x1 + 6.0, y1 + 4.0), leader)
            if overlap == 0.0 and not leader:
                break

    assert best is not None
    occupied.append(best[2])
    return best

class MasterCartographicPlate:
    """Builds a 3840x2160 georeferenced 1914 Western Europe master relief plate by fusing
    the two clean Natural Earth SR_HR frame-0 plates (1000 km & 530 km), multi-octave
    micro-relief, aged parchment grain, and bathymetric sea shading."""

    def __init__(self, clip_sh001_mp4: str, clip_sh002_mp4: str):
        self.master_view = View(4.95, 49.31, 1000.0, 3840, 2160)
        self.image = self._build_plate(clip_sh001_mp4, clip_sh002_mp4)

    def _read_f0_rgb(self, mp4_path: str) -> np.ndarray:
        cap = cv2.VideoCapture(mp4_path)
        ok, frame = cap.read()
        cap.release()
        if not ok:
            raise RuntimeError(f"Cannot read frame 0 from {mp4_path}")
        return cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)

    def _build_plate(self, clip_sh001_mp4: str, clip_sh002_mp4: str) -> Image.Image:
        pw, ph = self.master_view.width, self.master_view.height
        f0_1000 = self._read_f0_rgb(clip_sh002_mp4)
        f0_530 = self._read_f0_rgb(clip_sh001_mp4)

        base_rgb = cv2.resize(f0_1000, (pw, ph), interpolation=cv2.INTER_CUBIC).astype(np.float32)

        # Project 530 km plate into master_view coordinates and cosine-feather blend (zero NaN edge)
        v1 = View(4.84, 49.90, 530.0, 1920, 1080)
        x0, y0 = self.master_view.to_px(*v1.to_lonlat(0, 0))
        x1, y1 = self.master_view.to_px(*v1.to_lonlat(1920, 1080))
        ix0, iy0, ix1, iy1 = int(round(x0)), int(round(y0)), int(round(x1)), int(round(y1))
        patch_530 = cv2.resize(f0_530, (ix1 - ix0, iy1 - iy0), interpolation=cv2.INTER_CUBIC).astype(np.float32)

        wx = np.clip(np.sin(np.linspace(0.0, math.pi, ix1 - ix0, dtype=np.float64)), 0.0, 1.0) ** 0.45
        wy = np.clip(np.sin(np.linspace(0.0, math.pi, iy1 - iy0, dtype=np.float64)), 0.0, 1.0) ** 0.45
        w2d = np.clip((wy[:, None] * wx[None, :]) * 1.35, 0.0, 1.0).astype(np.float32)[..., None]
        base_rgb[iy0:iy1, ix0:ix1] = base_rgb[iy0:iy1, ix0:ix1] * (1.0 - w2d) + patch_530 * w2d

        # Segment sea vs land from blue-red channel difference on the clean plate
        sea_score = (base_rgb[:, :, 2] - base_rgb[:, :, 0])
        is_sea = (sea_score > 6.0).astype(np.float32)
        sea_soft = cv2.GaussianBlur(is_sea, (0, 0), 3.0)

        # Extract relief luminance and synthesize fine micro-topography so Flanders zoom is crisp
        luma = (0.299 * base_rgb[:, :, 0] + 0.587 * base_rgb[:, :, 1] + 0.114 * base_rgb[:, :, 2]) / 255.0
        rng = np.random.default_rng(19141225)
        n_fine = cv2.GaussianBlur(rng.standard_normal((ph, pw)).astype(np.float32), (0, 0), 1.5)
        n_med = cv2.GaussianBlur(rng.standard_normal((ph, pw)).astype(np.float32), (0, 0), 4.8)
        gx = cv2.Sobel(n_fine + 0.65 * n_med, cv2.CV_32F, 1, 0, ksize=3)
        gy = cv2.Sobel(n_fine + 0.65 * n_med, cv2.CV_32F, 0, 1, ksize=3)
        micro_hillshade = np.clip((gx * 0.7 - gy * 0.7) * 0.020, -0.05, 0.05)
        relief_gain = np.clip(1.0 + 1.35 * (luma - 0.86) + micro_hillshade * (1.0 - sea_soft), 0.58, 1.16)

        paper_grain = rng.standard_normal((ph, pw)).astype(np.float32) * 0.014
        land_col = np.array(PALETTE["paper"], dtype=np.float32)[None, None, :] * (relief_gain + paper_grain)[..., None]

        dist_from_land = cv2.distanceTransform((is_sea > 0.5).astype(np.uint8), cv2.DIST_L2, 5)
        shelf_t = np.exp(-dist_from_land / 85.0)[..., None]
        sea_deep = np.array(PALETTE["sea_deep"], dtype=np.float32)[None, None, :]
        sea_shelf = np.array(PALETTE["sea_shelf"], dtype=np.float32)[None, None, :]
        sea_col = (sea_deep * (1.0 - shelf_t) + sea_shelf * shelf_t) * (1.0 + paper_grain * 0.6)[..., None]

        river_mask = np.clip((sea_score + 4.0) / 22.0, 0.0, 1.0) * (1.0 - is_sea)
        river_col = np.array(PALETTE["river"], dtype=np.float32)[None, None, :]
        land_with_rivers = land_col * (1.0 - river_mask[..., None] * 0.75) + river_col * (river_mask[..., None] * 0.75)

        plate_arr = land_with_rivers * (1.0 - sea_soft[..., None]) + sea_col * sea_soft[..., None]
        return Image.fromarray(np.clip(plate_arr, 0, 255).astype(np.uint8), mode="RGB")

    def crop_for_view(self, v: View) -> Image.Image:
        x0, y0 = self.master_view.to_px(*v.to_lonlat(0, 0))
        x1, y1 = self.master_view.to_px(*v.to_lonlat(v.width, v.height))
        return self.image.transform(
            (v.width, v.height),
            Image.EXTENT,
            (float(x0), float(y0), float(x1), float(y1)),
            Image.BICUBIC,
        )

_FONT_CACHE: dict[tuple[str, int], ImageFont.FreeTypeFont] = {}


def get_font(weight: str = "SemiBold", size: int = 24) -> ImageFont.FreeTypeFont:
    key = (weight, int(size))
    if key not in _FONT_CACHE:
        candidates = [
            f"/usr/share/fonts/opentype/inter/Inter-{weight}.otf",
            f"/usr/share/fonts/opentype/inter/InterDisplay-{weight}.otf",
            "/usr/share/fonts/opentype/inter/Inter-Regular.otf",
        ]
        for path in candidates:
            if os.path.exists(path):
                _FONT_CACHE[key] = ImageFont.truetype(path, max(10, int(size)))
                break
        else:
            _FONT_CACHE[key] = ImageFont.load_default(size=max(10, int(size)))
    return _FONT_CACHE[key]


def draw_dashed_screen_line(draw: ImageDraw.ImageDraw, pts: list[tuple[float, float]], fill, width: int, on: float, off: float):
    period = on + off
    phase = 0.0
    for a, b in zip(pts, pts[1:]):
        seg_len = math.hypot(b[0] - a[0], b[1] - a[1])
        if seg_len <= 1e-4:
            continue
        pos = 0.0
        while pos < seg_len:
            step = (on - phase) if phase < on else (period - phase)
            end = min(seg_len, pos + max(step, 0.5))
            if phase < on:
                p0 = (a[0] + (b[0] - a[0]) * (pos / seg_len), a[1] + (b[1] - a[1]) * (pos / seg_len))
                p1 = (a[0] + (b[0] - a[0]) * (end / seg_len), a[1] + (b[1] - a[1]) * (end / seg_len))
                draw.line([p0, p1], fill=fill, width=width)
            phase = (phase + (end - pos)) % period
            pos = end


def draw_arrowhead(draw: ImageDraw.ImageDraw, tip: tuple[float, float], back: tuple[float, float], size: float, fill, outline):
    dx, dy = tip[0] - back[0], tip[1] - back[1]
    norm = math.hypot(dx, dy) or 1.0
    ux, uy = dx / norm, dy / norm
    left = (tip[0] - ux * size - uy * size * 0.58, tip[1] - uy * size + ux * size * 0.58)
    right = (tip[0] - ux * size + uy * size * 0.58, tip[1] - uy * size - ux * size * 0.58)
    wing_in = (tip[0] - ux * size * 0.72, tip[1] - uy * size * 0.72)
    poly = [tip, left, wing_in, right]
    draw.polygon(poly, fill=fill)
    draw.line(poly + [tip], fill=outline, width=max(2, int(size * 0.12)), joint="curve")


def apply_clean_pitch_warp(im: Image.Image, pitch_deg: float) -> Image.Image:
    """Applies perspective pitch and crops to the inscribed rectangle so zero black corner wedges appear."""
    if pitch_deg <= 0.5:
        return im
    w, h = im.size
    squeeze = 0.24 * math.sin(math.radians(pitch_deg))
    inset = squeeze * w * 0.5
    # Overscale horizontally by (1 + squeeze) before QUAD warp so top corners remain 100% filled with map pixels
    pad_x = int(math.ceil(inset)) + 4
    expanded = im.resize((w + 2 * pad_x, h), Image.BICUBIC)
    quad = (0.0, 0.0, float(pad_x), float(h), float(w + pad_x), float(h), float(w + 2 * pad_x), 0.0)
    return expanded.transform((w, h), Image.QUAD, quad, Image.BICUBIC)

# 6. AUDIO CONCATENATION & FRAME-ACCURATE SOUND CUE MIXING
# =============================================================================

def build_continuous_audio(wav_paths: Sequence[str], out_wav_path: str) -> None:
    """Concatenate the 3 real narrator audio tracks (S02_SH001..003) and layer subtle
    cartographic foley cues + 0.50 s trench-wind/cough J-cut prelap at exact word frames."""
    segments = []
    sr = 24000
    for idx, wp in enumerate(wav_paths):
        with wave.open(wp, "rb") as wf:
            sr = wf.getframerate()
            raw = np.frombuffer(wf.readframes(wf.getnframes()), dtype=np.int16).astype(np.float32) / 32768.0
        target_f = (133, 188, 132)[idx]
        target_samples = int(round(target_f * sr / FPS))
        if len(raw) >= target_samples:
            seg = raw[:target_samples]
        else:
            seg = np.pad(raw, (0, target_samples - len(raw)))
        segments.append(seg)

    full_audio = np.concatenate(segments)
    n_total = len(full_audio)
    rng = np.random.default_rng(1914)

    def add_wood_tick(frame_idx: int, gain: float = 0.09, freq: float = 180.0):
        s0 = int(round(frame_idx * sr / FPS))
        dur = int(0.045 * sr)
        if 0 <= s0 < n_total - dur:
            t = np.linspace(0.0, 0.045, dur, dtype=np.float32)
            env = np.exp(-t * 95.0)
            click = (np.sin(2.0 * math.pi * freq * t) + 0.5 * np.sin(2.0 * math.pi * (freq * 2.4) * t)) * env * gain
            full_audio[s0:s0 + dur] += click

    def add_charcoal_draw(f_start: int, f_end: int, gain: float = 0.024):
        s0 = int(round(f_start * sr / FPS))
        s1 = min(n_total, int(round(f_end * sr / FPS)))
        if s1 > s0:
            noise = rng.standard_normal(s1 - s0).astype(np.float32)
            kernel = np.ones(18, dtype=np.float32) / 18.0
            filtered = np.convolve(noise, kernel, mode="same")
            env = np.clip(np.sin(np.linspace(0.0, math.pi, s1 - s0, dtype=np.float64)), 0.0, 1.0).astype(np.float32) ** 0.5
            full_audio[s0:s1] += filtered * env * gain

    add_wood_tick(36, gain=0.09, freq=195.0)       # "August" date card stamp (f36)
    add_charcoal_draw(76, 118, gain=0.022)         # German August advance draw (f76..118)
    add_wood_tick(118, gain=0.08, freq=165.0)      # Advance arrival at Marne (f118)
    add_wood_tick(149, gain=0.09, freq=190.0)      # "December" date card flip (f149)
    add_charcoal_draw(231, 308, gain=0.025)        # Western Front trench draw (f231..308)
    add_wood_tick(276, gain=0.07, freq=220.0)      # "North Sea" pin (f276)
    add_wood_tick(293, gain=0.08, freq=175.0)      # "Switzerland" pin (f293)
    add_wood_tick(409, gain=0.09, freq=210.0)      # "close" Flanders sector glow (f409)

    s_pre = int(round(440 * sr / FPS))
    if s_pre < n_total:
        wind = np.convolve(rng.standard_normal(n_total - s_pre).astype(np.float32), np.ones(45) / 45.0, mode="same")
        ramp = np.linspace(0.0, 0.045, n_total - s_pre, dtype=np.float32)
        full_audio[s_pre:] += wind * ramp

    full_audio = np.nan_to_num(full_audio, nan=0.0, posinf=0.0, neginf=0.0)
    stereo = np.stack([full_audio, full_audio], axis=1)
    pcm16 = np.clip(stereo * 32767.0, -32768, 32767).astype(np.int16)
    with wave.open(out_wav_path, "wb") as wf:
        wf.setnchannels(2)
        wf.setsampwidth(2)
        wf.setframerate(sr)
        wf.writeframes(pcm16.tobytes())

        [
            "ffmpeg", "-y", "-loglevel", "error",
            "-f", "rawvideo", "-vcodec", "rawvideo",
            "-pix_fmt", "rgb24", "-s", f"{WIDTH}x{HEIGHT}", "-r", str(FPS),
            "-i", "-",
            "-i", master_wav,

    if proc.wait(timeout=300) != 0:
